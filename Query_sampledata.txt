#Queries to sampledata
prefix mo: <https://purl.org/ontology/modalia#>
prefix dcterms: <https://purl.org/ontology/dc/terms/> 

SELECT ?s ?ttl ?ordr ?lr ?lrttl
{
    #?s ?p ?o
   ?s a mo:LearningPath .
   ?s dcterms:title ?ttl .
     ?stp mo:belongsToLearningPath ?s .
     ?stp mo:hasEducationalResource ?lr .
	 ?stp mo:hasOrder ?ordr .
     ?lr dcterms:title ?lrttl.
}
order by ?ordr
